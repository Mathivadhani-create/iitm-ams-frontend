  getCurrentUser: async () => {
    const res = await fetch(`${API_BASE}/auth/me`, {
      headers: getAuthHeaders(),
    });

    const data = await handleResponse<{
      user: User;
      student?: Student;
      faculty?: Faculty;
    }>(res);

    return {
      user: data.user,
      student: data.student,
      faculty: data.faculty,
    };
  },
